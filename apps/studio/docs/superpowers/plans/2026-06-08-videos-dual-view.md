# Videos 双视图重做 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 9-column text-only Videos page with a list/grid dual-view that introduces thumbnails, consolidates the "platform × status × phone" columns into a single badge unit, and lets users switch views with a persisted preference.

**Architecture:** Backend grows `thumbnail_path` + `duration_seconds` columns on `videos`, plus a `thumbnail_service` that shells out to `ffmpeg`/`ffprobe` on upload (silently degrades when unavailable). The `/api/videos` list endpoint gains `thumbnail_url`, `duration_seconds`, per-platform `phone_label`, and a `summary_counts` object. Frontend introduces five new reusable components (`VideoThumbnail`, `PlatformBadgeColumn`, `ViewSwitcher`, `DateSectionHead`, `VideoMediaRow` / `VideoCard`) consumed by two view shells (`VideoListView` virtualized, `VideoGridView` CSS-grid). View choice persists via `useVideoView` → `localStorage['videos:view']`.

**Tech Stack:** Python 3.11 + FastAPI + SQLModel + Alembic + Procrastinate (backend), React 19 + TypeScript + Tailwind + `@tanstack/react-virtual` + i18next (frontend), system `ffmpeg`/`ffprobe` (image/duration extraction).

**Spec:** [`docs/superpowers/specs/2026-06-08-videos-dual-view-design.md`](../specs/2026-06-08-videos-dual-view-design.md)

Create `app/services/thumbnail_service.py`:

```python
"""ffmpeg-driven thumbnail + duration extraction.

Used at upload time to capture a single frame around the 1-second
mark and the video's duration. Any failure (ffmpeg missing, bad
input, timeout) degrades silently to (None, None) — the frontend
has a placeholder. Never raise from this module.
"""
from __future__ import annotations

import logging
import subprocess
from pathlib import Path
from typing import Optional

from app import config

logger = logging.getLogger(__name__)

THUMBNAIL_DIR = config.DATA_DIR / "thumbnails"
TIMEOUT_SECONDS = 60
THUMB_WIDTH = 480
_LOGGED_MISSING_FFMPEG = False


def _ffmpeg_thumb(video: Path, out: Path) -> bool:
    """Run ffmpeg → JPEG. Returns True on success."""
    out.parent.mkdir(parents=True, exist_ok=True)
    cmd = [
        "ffmpeg", "-y",
        "-ss", "00:00:01",
        "-i", str(video),
        "-vframes", "1",
        "-vf", f"scale={THUMB_WIDTH}:-2",
        "-q:v", "4",
        str(out),
    ]
    try:
        r = subprocess.run(cmd, capture_output=True,
                           timeout=TIMEOUT_SECONDS)
    except (FileNotFoundError, subprocess.TimeoutExpired) as e:
        global _LOGGED_MISSING_FFMPEG
        if isinstance(e, FileNotFoundError) and not _LOGGED_MISSING_FFMPEG:
            logger.warning("ffmpeg not on PATH — thumbnails disabled")
            _LOGGED_MISSING_FFMPEG = True
        return False
    if r.returncode != 0:
        logger.info("ffmpeg failed video=%s stderr=%s",
                    video, r.stderr[-200:].decode(errors="replace"))
        return False
    return out.exists() and out.stat().st_size > 0


def _ffprobe_duration(video: Path) -> Optional[int]:
    """Probe duration → integer seconds, or None on any failure."""
    cmd = [
        "ffprobe", "-v", "error",
        "-show_entries", "format=duration",
        "-of", "default=nw=1:nk=1",
        str(video),
    ]
    try:
        r = subprocess.run(cmd, capture_output=True,
                           timeout=TIMEOUT_SECONDS)
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return None
    if r.returncode != 0:
        return None
    raw = r.stdout.decode(errors="replace").strip()
    try:
        return int(round(float(raw)))
    except ValueError:
        return None


def generate_thumbnail(
    video_path: Path | str, *, video_id: int,
) -> tuple[Optional[str], Optional[int]]:
    """Extract a JPEG thumbnail + duration. Never raises.

    Returns (relative_path, duration_seconds) where relative_path is
    'thumbnails/<id>.jpg' on success or None on any failure.
    duration_seconds may still be returned even when thumbnail
    extraction failed (e.g. ffprobe present, ffmpeg not).
    """
    video = Path(video_path)
    out = THUMBNAIL_DIR / f"{video_id}.jpg"
    ok = _ffmpeg_thumb(video, out)
    duration = _ffprobe_duration(video)
    rel = f"thumbnails/{video_id}.jpg" if ok else None
    return rel, duration
```

```python
def summary_counts_for(
    session: Session, *, group_id: int | str,
) -> dict:
    """Counts for the page-level summary line.

    total   – number of videos in the current group
    success – videos where every publication is SUCCESS
    pending – videos with ≥1 non-terminal publication
    """
    gfrag, gparams = _group_filter_sql(group_id)
    sql = f"""
        WITH v_pub AS (
          SELECT v.id,
                 COUNT(p.id) AS n,
                 SUM(CASE WHEN p.result = 'SUCCESS' THEN 1 ELSE 0 END) AS ok,
                 SUM(CASE WHEN p.result NOT IN ('SUCCESS','FAILURE')
                          THEN 1 ELSE 0 END) AS not_terminal
            FROM videos v
            LEFT JOIN publications p ON p.video_id = v.id
            WHERE {gfrag}
            GROUP BY v.id
        )
        SELECT
          COUNT(*)                                        AS total,
          SUM(CASE WHEN n > 0 AND ok = n THEN 1 ELSE 0 END) AS success,
          SUM(CASE WHEN n = 0 OR not_terminal > 0 THEN 1 ELSE 0 END) AS pending
        FROM v_pub
    """
    row = session.execute(sa.text(sql), gparams).first()
    if row is None:
        return {"total": 0, "success": 0, "pending": 0}
    return {
        "total": int(row[0] or 0),
        "success": int(row[1] or 0),
        "pending": int(row[2] or 0),
    }
```

```python
import logging
from pathlib import Path

from app.db import Session, engine
from app.models import Video
from app.repositories import videos as vid_repo
from app.services.thumbnail_service import generate_thumbnail

@app.task(name="backfill_thumbnails", queue="default")

            session.query(Video.id, Video.stored_path)
```

```typescript
import { useEffect, useMemo, useRef } from 'react'
import { useTranslation } from 'react-i18next'
import { useVirtualizer } from '@tanstack/react-virtual'
import type { VideoListRow } from '../../lib/useVideoPages'
import VideoMediaRow from './VideoMediaRow'
import { DateSectionHead } from './DateSectionHead'
import VideoDetailPanel from './VideoDetailPanel'
import { parseBackendIso } from '../../lib/time'

type Row =
  | { kind: 'section'; iso: string | null; count: number }
  | { kind: 'video'; video: VideoListRow }
  | { kind: 'detail'; videoId: number }
  | { kind: 'sentinel' }

function dateKey(iso: string | null): string {
  const d = parseBackendIso(iso ?? '')
  if (!d) return 'unknown'
  return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`
}

type Props = {
  items: VideoListRow[]
  hasMore: boolean
  loading: boolean
  expandedIds: Set<number>
  selectedIds: Set<number>
  groupsById: Map<number, string>
  onRowClick: (id: number, mods: { shift: boolean; meta: boolean }) => void
  onToggleExpanded: (id: number) => void
  onDelete: (id: number) => void
  onLoadMore: () => void
  onOpenFailureShot: (url: string, platform: string, detail: string) => void
}

export function VideoListView({
  items, hasMore, loading, expandedIds, selectedIds, groupsById,
  onRowClick, onToggleExpanded, onDelete, onLoadMore, onOpenFailureShot,
}: Props) {
  const { t } = useTranslation()
  const containerRef = useRef<HTMLDivElement | null>(null)

  const rows: Row[] = useMemo(() => {
    const out: Row[] = []
    const sectionCounts: number[] = []
    let prevKey: string | null = null
    let curSectionRef: { count: number } | null = null
    for (const v of items) {
      const k = dateKey(v.uploaded_at)
      if (k !== prevKey) {
        if (curSectionRef) sectionCounts.push(curSectionRef.count)
        curSectionRef = { count: 0 }
        out.push({ kind: 'section', iso: v.uploaded_at, count: 0 })
        prevKey = k
      }
      out.push({ kind: 'video', video: v })
      if (expandedIds.has(v.id)) out.push({ kind: 'detail', videoId: v.id })
      if (curSectionRef) curSectionRef.count += 1
    }
    // Fill the count back into each section placeholder
    let cursor = 0
    let segIdx = 0
    if (curSectionRef) sectionCounts.push(curSectionRef.count)
    for (let i = 0; i < out.length; i++) {
      const r = out[i]
      if (r.kind === 'section') {
        out[i] = { kind: 'section', iso: r.iso, count: sectionCounts[segIdx] ?? 0 }
        segIdx += 1
      }
      cursor += 1
    }
    out.push({ kind: 'sentinel' })
    return out
  }, [items, expandedIds])

  const virtualizer = useVirtualizer({
    count: rows.length,
    getScrollElement: () => containerRef.current,
    estimateSize: (i) => {
      const r = rows[i]
      if (r.kind === 'section') return 32
      if (r.kind === 'detail') return 180
      if (r.kind === 'sentinel') return 40
      return 64
    },
    overscan: 6,
    measureElement: (el) => el.getBoundingClientRect().height,
  })

  const endIndex = virtualizer.range?.endIndex ?? null
  useEffect(() => {
    if (!hasMore || loading) return
    if (endIndex !== null && endIndex >= rows.length - 1) onLoadMore()
  }, [endIndex, rows.length, hasMore, loading, onLoadMore])

  return (
    <div
      ref={containerRef}
      role="table"
      className="border border-[var(--color-line)] rounded-[10px] bg-[var(--color-surface)] shadow-[0_1px_2px_rgba(0,0,0,0.04)] overflow-auto max-h-[calc(100vh-260px)]"
    >
      <div
        role="rowgroup"
        style={{ height: `${virtualizer.getTotalSize()}px`, position: 'relative' }}
      >
        {virtualizer.getVirtualItems().map((vi) => {
          const row = rows[vi.index]
          return (
            <div
              key={vi.key}
              data-index={vi.index}
              ref={virtualizer.measureElement}
              style={{
                position: 'absolute', top: 0, left: 0, right: 0,
                transform: `translateY(${vi.start}px)`,
              }}
            >
              {row.kind === 'section' && (
                <DateSectionHead iso={row.iso} count={row.count} />
              )}
              {row.kind === 'video' && (
                <VideoMediaRow
                  row={row.video}
                  groupsById={groupsById}
                  isExpanded={expandedIds.has(row.video.id)}
                  isSelected={selectedIds.has(row.video.id)}
                  onRowClick={onRowClick}
                  onToggleExpanded={onToggleExpanded}
                  onDelete={onDelete}
                />
              )}
              {row.kind === 'detail' && (
                <VideoDetailPanel
                  videoId={row.videoId}
                  onOpenFailureShot={onOpenFailureShot}
                />
              )}
              {row.kind === 'sentinel' && (
                <div className="px-3 py-3 text-center text-[12px] text-[var(--color-ink-5)]">
                  {hasMore
                    ? (loading
                        ? t('videos.loadingMore', { defaultValue: '加载中…' })
                        : t('videos.scrollForMore', { defaultValue: '滚动加载更多' }))
                    : t('videos.endOfList', { defaultValue: '— 已到底部 —' })}
                </div>
              )}
            </div>
          )
        })}
      </div>
    </div>
  )
}
```

```typescript
import { useEffect, useMemo, useRef } from 'react'
import { useTranslation } from 'react-i18next'
import type { VideoListRow } from '../../lib/useVideoPages'
import { VideoCard } from './VideoCard'
import { DateSectionHead } from './DateSectionHead'
import { parseBackendIso } from '../../lib/time'

function dateKey(iso: string | null): string {
  const d = parseBackendIso(iso ?? '')
  if (!d) return 'unknown'
  return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`
}

type Props = {
  items: VideoListRow[]
  hasMore: boolean
  loading: boolean
  selectedIds: Set<number>
  groupsById: Map<number, string>
  onCardClick: (id: number, mods: { shift: boolean; meta: boolean }) => void
  onDelete: (id: number) => void
  onOpenDetail: (id: number) => void
  onLoadMore: () => void
}

export function VideoGridView({
  items, hasMore, loading, selectedIds, groupsById,
  onCardClick, onDelete, onOpenDetail, onLoadMore,
}: Props) {
  const { t } = useTranslation()
  const sections = useMemo(() => {
    const map = new Map<string, { iso: string | null; rows: VideoListRow[] }>()
    for (const v of items) {
      const k = dateKey(v.uploaded_at)
      const cur = map.get(k)
      if (cur) cur.rows.push(v)
      else map.set(k, { iso: v.uploaded_at, rows: [v] })
    }
    return Array.from(map.values())
  }, [items])

  // Infinite-scroll sentinel
  const sentinelRef = useRef<HTMLDivElement | null>(null)
  useEffect(() => {
    if (!sentinelRef.current) return
    const obs = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting && hasMore && !loading) onLoadMore()
    }, { rootMargin: '200px' })
    obs.observe(sentinelRef.current)
    return () => obs.disconnect()
  }, [hasMore, loading, onLoadMore])

  return (
    <div className="rounded-[10px] bg-[var(--color-surface)] border border-[var(--color-line)] shadow-[0_1px_2px_rgba(0,0,0,0.04)] overflow-hidden">
      {sections.map((sec, idx) => (
        <div key={idx}>
          <DateSectionHead iso={sec.iso} count={sec.rows.length} />
          <div
            className="grid gap-3 p-3"
            style={{ gridTemplateColumns: 'repeat(auto-fill, minmax(220px, 1fr))' }}
          >
            {sec.rows.map((row) => (
              <VideoCard
                key={row.id}
                row={row}
                groupsById={groupsById}
                isSelected={selectedIds.has(row.id)}
                onClick={onCardClick}
                onDelete={onDelete}
                onOpenDetail={onOpenDetail}
              />
            ))}
          </div>
        </div>
      ))}
      <div ref={sentinelRef} className="h-10 flex items-center justify-center text-[12px] text-[var(--color-ink-5)]">
        {hasMore
          ? (loading
              ? t('videos.loadingMore', { defaultValue: '加载中…' })
              : t('videos.scrollForMore', { defaultValue: '滚动加载更多' }))
          : t('videos.endOfList', { defaultValue: '— 已到底部 —' })}
      </div>
    </div>
  )
}
```
